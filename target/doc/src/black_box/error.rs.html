9/// Enum representing the type of error that occurred.
10/// Each variant can be extended to capture different categories of failures,
11/// such as system-level, I/O, validation, or network-related issues.
12///
13/// # Author
14/// K Vishwanath
15#[derive(Debug)]
16pub enum ErrorType {
17    /// Represents a generic system-level error (e.g., environment or I/O failures).
18    SystemError,
19    // Future variants:
20    // IoError,
21    // ValidationError,
22    // NetworkError,
23}

25/// Represents an application-level error with a message and its category.
26///
27/// This struct encapsulates both the descriptive message and
28/// a high-level classification of the error (via [`ErrorType`]).
29///
30/// # Author
31/// K Vishwanath
32#[derive(Debug)]
33pub struct AppError {
34    /// The detailed error message.
35    message: String,
36
37    /// The type/category of error (see [`ErrorType`]).
38    error_type: ErrorType,
39}
40
41/// Implementation for AppError.
42///
43/// # Author
44/// K Vishwanath
45impl AppError {
46    /// Creates a new [`AppError`] instance.
47    ///
48    /// # Arguments
49    ///
50    /// * `message` - A descriptive message about what went wrong.
51    /// * `error_type` - The specific type of the error.
52    ///
53    /// # Example
54    /// ```
55    /// use crate::error::{AppError, ErrorType};
56    /// let err = AppError::new("Failed to open file", ErrorType::SystemError);
57    /// ```
58    pub fn new(message: impl Into<String>, error_type: ErrorType) -> Self {
59        Self {
60            message: message.into(),
61            error_type,
62        }
63    }
64
65    /// Returns the descriptive error message.
66    pub fn get_message(&self) -> &str {
67        &self.message
68    }
69
70    /// Returns the associated [`ErrorType`] of this error.
71    pub fn get_error_type(&self) -> &ErrorType {
72        &self.error_type
73    }
74}

76/// Implements [`Display`] for [`AppError`], enabling user-friendly formatting.
77///
78/// If your IDE (e.g., RustRover) highlights `Display` in red, this can be ignored.
79/// The implementation is correct and functional.
80///
81/// # Author
82/// K Vishwanath
83impl Display for AppError {
84    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
85        write!(f, "{:?}: {}", self.error_type, self.message)
86    }
